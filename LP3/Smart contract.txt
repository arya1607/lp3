How to Run the Program
1. Open Remix IDE.
2. Create a file named Bank.sol.
3. Paste the Solidity code.
4. Open the Solidity Compiler.
5. Compile using Solidity 0.8.x.
6. Open the Deploy and Run Transactions tab.
7. Select Remix VM or connect MetaMask to Sepolia.
8. Deploy the contract.
9. Enter an amount in the Value field and click deposit.
10. Click showBalance to view the balance.
11. Enter the withdrawal amount in wei and click withdraw.


// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract Bank {

    mapping(address => uint256) private balances;

    // Deposit Ether
    function deposit() public payable {

        require(msg.value > 0, "Send some ether");

        balances[msg.sender] += msg.value;
    }

    // Withdraw Ether
    function withdraw(uint256 amount) public {

        require(
            balances[msg.sender] >= amount,
            "Insufficient balance"
        );

        balances[msg.sender] -= amount;

        payable(msg.sender).transfer(amount);
    }

    // Show balance
    function showBalance() public view returns (uint256) {

        return balances[msg.sender];
    }
}
